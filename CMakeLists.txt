cmake_minimum_required(VERSION 3.18)
project(aec48 C CXX)
set(CMAKE_CXX_STANDARD 17)
# 编译产物里不留本机绝对路径（assert / __FILE__ 会把源文件路径写进二进制）
add_compile_options(-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
add_library(pffft STATIC third_party/pffft/src/pffft.c third_party/pffft/src/pffft_common.c)
target_include_directories(pffft PUBLIC third_party/pffft/include/pffft)
set_target_properties(pffft PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND (ANDROID_ABI MATCHES "arm"))
  target_compile_definitions(pffft PRIVATE PFFFT_ENABLE_NEON)
endif()
target_compile_options(pffft PRIVATE -O3 -ffast-math)
# Arm KleidiAI（Apache-2.0，节选见 third_party/kleidiai/README.md）：掩码网络 CPF 两个全连接的 int8 量化打包与点积内核。
# 不加 -ffast-math：量化的取整与比例必须在各平台算得一样。点积内核只在 arm64 编，单独开 dotprod 指令（运行时确认 CPU 支持才调用）。
set(KAI third_party/kleidiai)
set(KAI_MM ${KAI}/kai/ukernels/matmul)
add_library(kleidiai STATIC ${KAI_MM}/pack/kai_lhs_quant_pack_qai8dxp_f32.c)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR (NOT ANDROID AND CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64"))
  set(KAI_DOT ${KAI_MM}/matmul_clamp_f32_qai8dxp_qsi8cxp/kai_matmul_clamp_f32_qai8dxp1x8_qsi8cxp4x8_1x4_neon_dotprod.c)
  target_sources(kleidiai PRIVATE ${KAI_MM}/pack/kai_rhs_pack_nxk_qsi8cxp_qsi8cx_neon.c ${KAI_DOT})
  set_source_files_properties(${KAI_DOT} PROPERTIES COMPILE_OPTIONS "-march=armv8.2-a+dotprod")
endif()
target_include_directories(kleidiai PUBLIC ${KAI})
target_compile_options(kleidiai PRIVATE -O3)
target_compile_definitions(kleidiai PRIVATE NDEBUG)
set_target_properties(kleidiai PROPERTIES POSITION_INDEPENDENT_CODE ON)

# 内嵌模型：默认 models.cpp；候选 models_v62b.cpp；讲话 models_talk.cpp；双档 models_dual.cpp。
# 父工程先 set(AEC48_MODELS …) 再 add_subdirectory 时保留它（cmake_minimum_required 3.18 下 CMP0126 为 OLD，直接建同名缓存项会把普通变量清掉）
if(NOT DEFINED AEC48_MODELS)
  set(AEC48_MODELS src/models.cpp CACHE STRING "内嵌模型源文件（tools/embed_models.py 生成）")
endif()
add_library(aec48 SHARED src/aec48.cpp ${AEC48_MODELS})
get_filename_component(AEC48_MODELS_NAME "${AEC48_MODELS}" NAME)
if(AEC48_MODELS_NAME STREQUAL "models_dual.cpp")
  target_compile_definitions(aec48 PRIVATE AEC48_DUAL)
endif()
target_include_directories(aec48 PUBLIC include PRIVATE src)
# 与端上核对版 chain_d 同编译选项：-O3 -ffast-math，时延估计用 float
target_compile_options(aec48 PRIVATE -O3 -ffast-math -fvisibility=hidden)
target_compile_definitions(aec48 PRIVATE NLIN_TDE_T=float USE_PFFFT COFI_Q8)
target_link_libraries(aec48 PRIVATE pffft kleidiai)
if(ANDROID)
  target_link_libraries(aec48 PRIVATE log)
  target_link_options(aec48 PRIVATE -Wl,--gc-sections -s)
endif()
# 自检程序（主机默认编；Android 加 -DAEC48_BUILD_CHECK=ON 也编，用来在手机上跑同一份自检）
if(AEC48_BUILD_CHECK OR NOT ANDROID)
  add_executable(aec48_check test/aec48_check.cpp)
  target_link_libraries(aec48_check PRIVATE aec48)
  add_executable(aec48_bench test/aec48_bench.cpp)
  target_link_libraries(aec48_bench PRIVATE aec48)
  add_executable(aec48_robust test/aec48_robust.cpp)
  target_link_libraries(aec48_robust PRIVATE aec48)
  add_executable(aec48_hook test/aec48_hook.cpp)
  target_link_libraries(aec48_hook PRIVATE aec48)
  add_executable(aec48_hook2 test/aec48_hook2.cpp)
  target_link_libraries(aec48_hook2 PRIVATE aec48)
  add_executable(aec48_hint test/aec48_hint.cpp)
  target_include_directories(aec48_hint PRIVATE src)
  target_compile_options(aec48_hint PRIVATE -O3 -ffast-math)
  add_executable(aec48_talk_gate test/aec48_talk_gate.cpp)
  target_include_directories(aec48_talk_gate PRIVATE src)
  target_compile_options(aec48_talk_gate PRIVATE -O3)
  add_executable(aec48_talk_api test/aec48_talk_api.cpp)
  target_link_libraries(aec48_talk_api PRIVATE aec48)
  add_executable(aec48_hpf test/aec48_hpf.cpp)
  target_include_directories(aec48_hpf PRIVATE src)
  target_compile_options(aec48_hpf PRIVATE -O3 -ffast-math)
  target_link_libraries(aec48_hpf PRIVATE aec48)
  add_executable(aec48_profile test/aec48_profile.cpp)
  target_link_libraries(aec48_profile PRIVATE aec48)
  # 私有观测口只进自检构建，不进对外头文件或发布库。
  option(AEC48_TESTING "启用频谱提示的私有自检观测口" OFF)
  if(AEC48_TESTING)
    target_compile_definitions(aec48 PRIVATE AEC48_TESTING)
    target_compile_definitions(aec48_hook2 PRIVATE AEC48_TESTING)
  endif()
  if(NOT ANDROID)
    target_link_libraries(aec48_robust PRIVATE pthread)
    target_link_libraries(aec48_profile PRIVATE pthread)
    # 内存分配失败注入：借 glibc 的 __libc_malloc 转发，只在能链接到它的 Linux 主机上编（macOS、musl 跳过）
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      include(CheckCSourceCompiles)
      check_c_source_compiles("#include <stddef.h>
extern void* __libc_malloc(size_t);
int main(void) { return __libc_malloc(1) == 0; }" AEC48_HAVE_LIBC_MALLOC)
      if(AEC48_HAVE_LIBC_MALLOC)
        add_executable(aec48_oom test/aec48_oom.cpp)
        target_link_libraries(aec48_oom PRIVATE aec48)
      endif()
    endif()
  endif()
endif()
